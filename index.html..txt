    <script>
        let currentUser = null;
        let inventario = [
            { id: '1', nombre: 'Vestido Satén', variante: 'S / Negro', precio: 35.00, stock: 8 },
            { id: '2', nombre: 'Vestido Satén', variante: 'M / Rosa Soft', precio: 35.00, stock: 5 },
            { id: '3', nombre: 'Blusa Casual', variante: 'S / Blanco', precio: 20.00, stock: 12 },
            { id: '4', nombre: 'Jeans Tiro Alto', variante: '32 / Azul Oscuro', precio: 45.00, stock: 7 }
        ];

        let pedidoActual = [];
        let productoSeleccionadoPOS = null;
        
        // Acumuladores de caja
        let statsCaja = { ventas: 0, efectivo: 0, zelle: 0, pagomovil: 0, cashea: 0, total: 0 };

        function login() {
            let u = document.getElementById('user-input').value.trim().toLowerCase();
            let p = document.getElementById('pass-input').value.trim();

            if((u === "admin" && p === "1234") || (u === "cajero" && p === "1234")) {
                currentUser = u;
                document.getElementById('login-screen').classList.add('hidden');
                document.getElementById('app-screen').classList.remove('hidden');
                document.getElementById('user-role-label').innerText = `Rol: ${currentUser.toUpperCase()}`;
                
                if(currentUser === 'cajero') {
                    document.getElementById('seccion-agregar-prod').style.display = 'none';
                }

                renderizarPOS();
                renderizarInventario();
                document.getElementById('fecha-hoy').innerText = new Date().toLocaleDateString();
            } else {
                alert("Credenciales incorrectas. Usa admin/1234 o cajero/1234");
            }
        }

        function switchTab(tabName) {
            document.querySelectorAll('.panel-content').forEach(el => el.classList.add('hidden'));
            document.querySelectorAll('.tab-btn').forEach(el => el.classList.remove('active'));
            
            if(tabName === 'pos') {
                document.getElementById('tab-pos').classList.remove('hidden');
                event.target.classList.add('active');
            } else if(tabName === 'inventario') {
                document.getElementById('tab-inventario').classList.remove('hidden');
                event.target.classList.add('active');
            } else if(tabName === 'caja') {
                document.getElementById('tab-caja').classList.remove('hidden');
                event.target.classList.add('active');
            }
        }

        function renderizarPOS(filtro = '') {
            let contenedor = document.getElementById('lista-productos-pos');
            contenedor.innerHTML = '';
            productoSeleccionadoPOS = null;
            let texto = filtro.toLowerCase();

            inventario.forEach(item => {
                let infoCompleta = `${item.nombre} ${item.variante}`.toLowerCase();
                if(infoCompleta.includes(texto)) {
                    contenedor.innerHTML += `
                        <div class="producto-card" onclick="seleccionarPOS('${item.id}', this)">
                            <div>
                                <strong style="font-size: 0.95rem; color: var(--text-main);">${item.nombre}</strong>
                                <div style="font-size: 0.8rem; color: var(--muted-pink); margin-top:2px;">${item.variante}</div>
                                <div style="font-size: 0.75rem; color: var(--text-muted); margin-top:4px;">Stock: ${item.stock}</div>
                            </div>
                            <div style="font-size: 1.05rem; font-weight: bold; color: var(--soft-pink); margin-top: 10px;">$${item.precio.toFixed(2)}</div>
                        </div>
                    `;
                }
            });
        }

        function filtrarProductos() {
            let texto = document.getElementById('buscador-productos').value;
            renderizarPOS(texto);
        }

        function seleccionarPOS(id, elemento) {
            document.querySelectorAll('.producto-card').forEach(el => el.classList.remove('selected'));
            elemento.classList.add('selected');
            productoSeleccionadoPOS = id;
        }

        function agregarAlPedido() {
            if(!productoSeleccionadoPOS) {
                alert("Selecciona un producto primero.");
                return;
            }
            let cant = parseInt(document.getElementById('cantidad-input').value);
            if(isNaN(cant) || cant <= 0) return;

            let item = inventario.find(i => i.id === productoSeleccionadoPOS);
            if(item.stock < cant) {
                alert("Stock insuficiente para esta cantidad.");
                return;
            }

            item.stock -= cant;

            let existe = pedidoActual.find(p => p.id === item.id);
            if(existe) {
                existe.cantidad += cant;
            } else {
                pedidoActual.push({ id: item.id, nombre: `${item.nombre} (${item.variante})`, precio: item.precio, cantidad: cant });
            }

            document.getElementById('buscador-productos').value = '';
            renderizarPOS();
            renderizarPedido();
            renderizarInventario();
        }

        function renderizarPedido() {
            let contenedor = document.getElementById('ticket-actual');
            let totalDiv = document.getElementById('total-pagar');
            let tasa = parseFloat(document.getElementById('tasa-cambio').value) || 1;
            let desc = parseFloat(document.getElementById('descuento-input').value) || 0;

            if(pedidoActual.length === 0) {
                contenedor.innerHTML = '<p style="color: var(--text-muted); font-size: 0.9rem;">No hay productos agregados.</p>';
                totalDiv.innerText = "Total: $0.00";
                return;
            }

            let html = "<ul style='padding-left: 15px;'>";
            let subtotalUSD = 0;
            pedidoActual.forEach(item => {
                let sub = item.precio * item.cantidad;
                subtotalUSD += sub;
                html += `<li style="font-size: 0.85rem; margin-bottom: 4px;">${item.cantidad}x ${item.nombre} - $${sub.toFixed(2)}</li>`;
            });
            html += "</ul>";
            contenedor.innerHTML = html;

            let totalUSD = Math.max(0, subtotalUSD - desc);
            let totalBs = totalUSD * tasa;
            totalDiv.innerHTML = `Total: $${totalUSD.toFixed(2)} USD <br><span style="font-size: 0.85rem; color: var(--text-muted);">Bs. ${totalBs.toFixed(2)}</span>`;
        }

        function calcularTotalActualUSD() {
            let subtotal = pedidoActual.reduce((sum, item) => sum + (item.precio * item.cantidad), 0);
            let desc = parseFloat(document.getElementById('descuento-input').value) || 0;
            return Math.max(0, subtotal - desc);
        }

        function abrirModalPago() {
            if(pedidoActual.length === 0) {
                alert("El ticket está vacío.");
                return;
            }
            let total = calcularTotalActualUSD();
            document.getElementById('modal-total-etiqueta').innerText = `Monto a Cobrar: $${total.toFixed(2)} USD`;
            document.getElementById('pago-efectivo').value = total.toFixed(2);
            document.getElementById('pago-zelle').value = 0;
            document.getElementById('pago-pagomovil').value = 0;
            document.getElementById('pago-cashea').value = 0;
            document.getElementById('modal-pago').classList.remove('hidden');
        }

        function cerrarModalPago() {
            document.getElementById('modal-pago').classList.add('hidden');
        }

        function confirmarPagoMixto() {
            let totalVenta = calcularTotalActualUSD();
            let efec = parseFloat(document.getElementById('pago-efectivo').value) || 0;
            let zell = parseFloat(document.getElementById('pago-zelle').value) || 0;
            let pmov = parseFloat(document.getElementById('pago-pagomovil').value) || 0;
            let cash = parseFloat(document.getElementById('pago-cashea').value) || 0;

            let sumaPagos = efec + zell + pmov + cash;
            if(Math.abs(sumaPagos - totalVenta) > 0.05) {
                alert(`La suma de los métodos ($${sumaPagos.toFixed(2)}) no coincide con el total de la venta ($${totalVenta.toFixed(2)}).`);
                return;
            }

            // Registrar estadísticas
            statsCaja.ventas += 1;
            statsCaja.efectivo += efec;
            statsCaja.zelle += zell;
            statsCaja.pagomovil += pmov;
            statsCaja.cashea += cash;
            statsCaja.total += totalVenta;

            actualizarVistaCaja();
            alert("¡Venta procesada y cobrada con éxito!");
            
            pedidoActual = [];
            document.getElementById('descuento-input').value = 0;
            cerrarModalPago();
            renderizarPedido();
        }

        function actualizarVistaCaja() {
            let tasa = parseFloat(document.getElementById('tasa-cambio').value) || 1;
            document.getElementById('cierre-cant-ventas').innerText = statsCaja.ventas;
            document.getElementById('cierre-efectivo').innerText = `$${statsCaja.efectivo.toFixed(2)}`;
            document.getElementById('cierre-zelle').innerText = `$${statsCaja.zelle.toFixed(2)}`;
            document.getElementById('cierre-pagomovil').innerText = `Bs. ${(statsCaja.pagomovil * tasa).toFixed(2)} ($${statsCaja.pagomovil.toFixed(2)})`;
            document.getElementById('cierre-cashea').innerText = `$${statsCaja.cashea.toFixed(2)}`;
            document.getElementById('cierre-total').innerText = `$${statsCaja.total.toFixed(2)}`;
        }

        function renderizarInventario() {
            let tbody = document.getElementById('tabla-inventario-body');
            tbody.innerHTML = '';
            inventario.forEach((item, index) => {
                let disabledAttr = (currentUser === 'cajero') ? 'disabled' : '';
                tbody.innerHTML += `
                    <tr>
                        <td>${item.nombre}</td>
                        <td>${item.variante}</td>
                        <td><input type="number" value="${item.precio}" step="0.1" ${disabledAttr} onchange="editarInv(${index}, 'precio', this.value)" style="width:80px; margin:0; padding:4px;"></td>
                        <td><input type="number" value="${item.stock}" ${disabledAttr} onchange="editarInv(${index}, 'stock', this.value)" style="width:70px; margin:0; padding:4px;"></td>
                        <td>${currentUser === 'admin' ? `<button onclick="eliminarInv(${index})" style="background:var(--danger); padding:4px 8px; font-size:0.75rem;">Eliminar</button>` : '<span style="color:var(--text-muted); font-size:0.8rem;">Solo lectura</span>'}</td>
                    </tr>
                `;
            });
            renderizarPOS();
        }

        function editarInv(index, campo, valor) {
            if(currentUser !== 'admin') {
                alert("Solo el Administrador puede modificar precios o stock directamente.");
                renderizarInventario();
                return;
            }
            inventario[index][campo] = parseFloat(valor);
            renderizarInventario();
        }

        function eliminarInv(index) {
            if(currentUser !== 'admin') return;
            inventario.splice(index, 1);
            renderizarInventario();
        }

        function agregarProductoNuevo() {
            if(currentUser !== 'admin') return;
            let nombre = document.getElementById('nuevo-nombre').value;
            let variante = document.getElementById('nuevo-variante').value;
            let precio = parseFloat(document.getElementById('nuevo-precio').value);
            let stock = parseInt(document.getElementById('nuevo-stock').value);

            if(!nombre || !variante || isNaN(precio) || isNaN(stock)) {
                alert("Completa todos los campos correctamente.");
                return;
            }

            let nuevoId = String(Date.now());
            inventario.push({ id: nuevoId, nombre, variante, precio, stock });

            document.getElementById('nuevo-nombre').value = '';
            document.getElementById('nuevo-variante').value = '';
            document.getElementById('nuevo-precio').value = '';
            document.getElementById('nuevo-stock').value = '10';
            renderizarInventario();
        }
